deve ser registrado aqui após carregar */ getModuleRegistry() { return { // FOUNDATION 'graphics-fallback': { global: 'SENTINEL_GRAPHICS_FALLBACK', type: 'config' }, 'bus': { global: 'SentinelBus', type: 'core' }, // KERNEL FOUNDATION 'core': { global: 'SentinelCore', type: 'kernel' }, 'state-machine': { global: 'SentinelStateMachine', type: 'kernel' }, 'kernel': { global: 'SentinelKernel', type: 'kernel' }, // RUNTIME CONTROL 'scheduler': { global: 'SentinelScheduler', type: 'kernel' }, 'memory': { global: 'SentinelMemory', type: 'kernel' }, // PROTOCOLS 'protocols': { global: 'SentinelBootKernel', type: 'kernel' } }; } /** * RESOLVER DE DEPENDÊNCIAS * Valida se todas as dependências de um módulo estão carregadas */ resolveDependencies(moduleName, moduleConfig) { if (!moduleConfig.depends || moduleConfig.depends.length === 0) { return { resolved: true, missing: [] }; } const missing = []; for (const dep of moduleConfig.depends) { if (!this.modules.has(dep) && !window[this.getModuleRegistry()[dep]?.global]) { missing.push(dep); } } return { resolved: missing.length === 0, missing }; } /** * LOAD MODULE * Carrega e registra um módulo no sistema */ async dependências const depCheck = this.resolveDependencies(moduleName, moduleConfig); if (!depCheck.resolved) { this.log(`[DEPENDENCY_ERROR] ${moduleName} requer: ${depCheck.missing.join(', ')}`, 'ERROR'); return false; } try { // Obtém a referência global do módulo const registry = this.getModuleRegistry()[moduleName]; if (!registry) { this.log(`[REGISTRY_ERROR] ${moduleName} não está no registro`, 'ERROR'); return false; } const moduleInstance = window[registry.global]; if (!moduleInstance) { this.log(`[LOAD_ERROR] ${moduleName} não foi carregado globalmente (${registry.global})`, 'ERROR'); return false; } // Registra módulo this.modules.set(moduleName, { name: moduleName, instance: moduleInstance, type: moduleConfig.type, initialized: false, loadTime: Date.now() }); this.stats.modulesLoaded++; this.log(`[LOADED] ${moduleName}`, 'SUCCESS'); return true; } catch (error) { this.log(`[EXCEPTION] ${moduleName}: ${error.message}`, 'ERROR'); this.stats.modulesFailed++; this.lastError = error; return false; } } /** * INITIALIZE MODULE * Inicializa um módulo (chama init() ou boot()) */ async initializeModule(moduleName) { const mod = this.modules.get(moduleName); if (!mod) { this.log(`[INIT_ERROR] ${moduleName} não foi carregado`, 'ERROR'); return false; } if (mod.initialized) { this.log(`[SKIP_INIT] ${moduleName} já foi inicializado`, 'SKIP'); return true; } try { const instance = mod.instance; // Tenta chamar init() if (typeof instance.init === 'function') { await instance.init(); this.log(`[INIT] ${moduleName}.init()`, 'INFO'); } // Tenta chamar boot() if (typeof instance.boot === 'function') { await instance.boot(); this.log(`[BOOT] ${moduleName}.boot()`, 'INFO'); } mod.initialized = true; this.stats.modulesInitialized++; // Emite evento de inicialização this.emit(`module:initialized`, { module: moduleName, ts: Date.now() }); return true; } catch (error) { this.log(`[INIT_EXCEPTION] ${moduleName}: ${error.message}`, 'ERROR'); this.stats.modulesFailed++; this.lastError = error; return false; } } /** * EXECUTE PHASE * Executa uma fase completa (load + init) */ async executePhase(phaseConfig) { this.currentPhase = this.loadModule(moduleConfig.name, moduleConfig); if (!loaded && moduleConfig.essential) { phaseFailed++; if (phaseConfig.critical) { this.log(`[CRITICAL] Falha crítica em ${moduleConfig.name}. Abortando boot.`, 'ERROR'); return false; } } else if (loaded) { phaseSuccess++; } } // INIT PHASE this.log(`\n[INIT] Inicializando ${phaseModules.length} módulos...`, 'PHASE'); for (const moduleConfig of phaseModules) { if (this.modules.has(moduleConfig.name)) { const initialized = await this.initializeModule(moduleConfig.name); if (!initialized && moduleConfig.essential) { phaseFailed++; } } } // SYNC PHASE this.log(`\n[SYNC] Sincronizando módulos da fase...`, 'PHASE'); await this.syncPhase(phaseConfig); // EMIT EVENT this.emit(`orchestrator:phase-complete`, { phase: phaseName, success: phaseSuccess, failed: phaseFailed, ts: Date.now() }); this.log(`\n[PHASE_OK] ${phaseName} completa (${phaseSuccess} OK, ${phaseFailed} FAIL)\n`, 'SUCCESS'); return true; } /** * SYNC PHASE * Sincronização entre módulos da mesma fase */ async syncPhase(phaseConfig) { for (const moduleConfig of phaseConfig.modules) { const mod = this.modules.get(moduleConfig.name); if (mod && typeof mod.instance.onPhaseSync === 'function') { try { await mod.instance.onPhaseSync(phaseConfig.phase); } catch (e) { this.log(`[SYNC_ERROR] ${moduleConfig.name}: ${e.message}`, 'WARN'); } } } } /** * EXECUTE BOOT * Executa sequência completa de boot */ async executeBoot() { if (this.isBootComplete) { this.log('[BOOT_SKIP] Sistema já foi iniciado', 'WARN'); return true; } this.stats.bootStartTime = Date.now(); this.log( '%c╔═══════════════════════════════════════════════════════════════╗\n║ SENTINEL VR-OS v9.0 — BOOT SEQUENCE INITIATED ║\n╚═══════════════════════════════════════════════════════════════╝', 'color:#7F00FF;font-weight:bold;font-size:14px;font-family:monospace;' );
